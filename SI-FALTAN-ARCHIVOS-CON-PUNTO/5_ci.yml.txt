name: CI

on:
  push:
    branches: [main]
  pull_request:

permissions:
  contents: read

# El proyecto vive en la carpeta nyor-store/ del repositorio.
defaults:
  run:
    working-directory: nyor-store

jobs:
  calidad:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: nyor-store/package-lock.json
      - run: npm ci
      - run: npm run format:check
      - run: npm run lint
      - run: npm run typecheck
      - run: npm test
      # Solo dependencias de producción: las herramientas de desarrollo tienen una
      # advertencia conocida de picomatch sin parche (ver docs/ESTADO.md).
      - run: npm audit --omit=dev --audit-level=high
      - run: npm run build

  e2e:
    runs-on: ubuntu-latest
    needs: calidad
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: nyor-store/package-lock.json
      - run: npm ci
      - run: npx playwright install --with-deps chrome
      - run: npm run test:e2e
